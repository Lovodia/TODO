import graphviz

erd = graphviz.Digraph('ERD', format='png')
erd.attr(rankdir='LR', fontsize='12')

# Таблица users
erd.node('users', '''<<TABLE BORDER="1" CELLBORDER="1" CELLSPACING="0">
  <TR><TD COLSPAN="2"><B>users</B></TD></TR>
  <TR><TD><B>id</B></TD><TD>UUID PK</TD></TR>
  <TR><TD>username</TD><TD>TEXT UNIQUE</TD></TR>
  <TR><TD>password</TD><TD>TEXT</TD></TR>
  <TR><TD>telegram_id</TD><TD>BIGINT</TD></TR>
  <TR><TD>created_at</TD><TD>TIMESTAMP</TD></TR>
</TABLE>>''')

# Таблица tasks
erd.node('tasks', '''<<TABLE BORDER="1" CELLBORDER="1" CELLSPACING="0">
  <TR><TD COLSPAN="2"><B>tasks</B></TD></TR>
  <TR><TD><B>id</B></TD><TD>UUID PK</TD></TR>
  <TR><TD>user_id</TD><TD>UUID FK → users.id</TD></TR>
  <TR><TD>title</TD><TD>TEXT</TD></TR>
  <TR><TD>completed</TD><TD>BOOLEAN</TD></TR>
  <TR><TD>created_at</TD><TD>TIMESTAMP</TD></TR>
  <TR><TD>updated_at</TD><TD>TIMESTAMP</TD></TR>
</TABLE>>''')

# Таблица two_fa_codes
erd.node('two_fa_codes', '''<<TABLE BORDER="1" CELLBORDER="1" CELLSPACING="0">
  <TR><TD COLSPAN="2"><B>two_fa_codes</B></TD></TR>
  <TR><TD><B>id</B></TD><TD>UUID PK</TD></TR>
  <TR><TD>user_id</TD><TD>UUID FK → users.id</TD></TR>
  <TR><TD>code</TD><TD>TEXT</TD></TR>
  <TR><TD>expires_at</TD><TD>TIMESTAMP</TD></TR>
  <TR><TD>created_at</TD><TD>TIMESTAMP</TD></TR>
</TABLE>>''')

# Связи
erd.edge('users', 'tasks', label='1:N', arrowhead='crow')
erd.edge('users', 'two_fa_codes', label='1:N', arrowhead='crow')

# Сохранить в PNG
erd.render(filename='todo_erd', cleanup=True)
